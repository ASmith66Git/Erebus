async performUDSHandshake(): Promise<boolean> {
  return new Promise(async (resolve, reject) => {
    // 1. Setup the listener FIRST
    const removeListener = await this.monitorCharacteristic(
      SHEARWATER_SERVICE_UUIDS[0],
      SHEARWATER_NOTIFY_CHAR,
      (base64Data) => {
        const hex = Buffer.from(base64Data, 'base64').toString('hex');
        // Check for 0x75 ACK in the UDS frame
        if (hex.includes('75')) {
          bleLog('HANDSHAKE', 'Received 0x75 ACK - Session Active!');
          removeListener();
          resolve(true);
        }
      }
    );

    // 2. Send the 0x35 packet
    const handshake = this.wrapUDSCommand(UDS.REQUEST_DOWNLOAD, [0x00, 0x34]);
    await this.writeCharacteristic(..., handshake);
    
    // 3. Add a timeout so it doesn't hang forever
    setTimeout(() => { removeListener(); reject('Handshake Timeout'); }, 5000);
  });
}
